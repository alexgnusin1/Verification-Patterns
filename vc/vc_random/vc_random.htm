VC_RANDOM

 

Overview

vc_random package contains the set of useful randomization functions.

 

 

Functions

Name

Arguments

Description

rn

Min:  integer

Max: integer

Returns the random value between Min and Max (inclusive).

rl

Min:  integer

Max: integer

Returns the random value between Min and Max (inclusive), with the more probablitiy given for the small numbers

rh

Min:  integer

Max: integer

Returns the random value between Min and Max (inclusive), with the more probablitiy given for the large numbers

rnc2 to  rnc5

Value1, weight1;

Value2, weight2

…..

Returns one of the specified input values. Uses Input values random weights to perform constraint-random choise.

 

 

 

 

 

Example1

 

module test ();

 

reg [0:31] seed;

integer i;

 

`include “vc_random.v”

 

initial begin

  rand.seed = 10;

  for (i=0; i<40; i=i+1)

    $display("rl = %0d  rn = %0d    rh = %0d",

      rl(0,20), rn(0,20), rh(0,20));

  $finish;

end

 

endmodule

 

 

 

 

 

 

 

 

 

 

 

 

Example2: Constraint-random FSM

 

 

module sender (clk, rst, b2s_ack, s2b_req);

input clk, rst;

input b2s_ack;

output s2b_req;

 

parameter IDLE = 0;

parameter BUSY = 1;

reg state;

 

`include "vc_random.v"

 

always @(posedge clk) begin

  if (rst) state <= IDLE;

  else begin

    if (state == IDLE && !b2s_ack) state <= rnc2(IDLE, 10-`SND_SPEED, BUSY, `SND_SPEED);

    if (state == BUSY && b2s_ack)  state <= rnc2(BUSY, 10-`SND_SPEED, IDLE, `SND_SPEED);

  end

end

 

assign s2b_req = (state == BUSY);

 

endmodule
